#### 2.1.6 メソッド

```typescript
// + ボタン押下時
function increaseLimit(): void {
  const newLimit = limit.value + 1;

  // 上限チェック
  if (newLimit > 5) {
    showToast("飲み過ぎです", "warning");
    return;
  }

  // 3本以上への増加時は確認
  if (newLimit >= 3) {
    pendingLimit.value = newLimit;
    confirmMessage.value = "本当に増やしますか？";
    showConfirmDialog.value = true;
  } else {
    applyLimitChange(newLimit);
  }
}

// - ボタン押下時
function decreaseLimit(): void {
  const newLimit = limit.value - 1;

  if (newLimit < 1) return;

  // 現在の飲酒本数が新しい目標を超える場合
  if (count.value > newLimit) {
    pendingLimit.value = newLimit;
    confirmMessage.value = `既に${count.value}本飲んでいますが、目標を${newLimit}本に設定しますか？`;
    showConfirmDialog.value = true;
  } else {
    applyLimitChange(newLimit);

    // 2本以下への減少時はポジティブフィードバック
    if (newLimit <= 2) {
      showToast("その調子！健康的な選択です！", "success");
    }
  }
}

// 確認ダイアログで「はい」選択時
function handleConfirm(): void {
  const newLimit = pendingLimit.value!;
  const oldLimit = limit.value;

  showConfirmDialog.value = false;
  applyLimitChange(newLimit);

  // 増加の場合：爆発エフェクト
  if (newLimit > oldLimit && newLimit >= 3) {
    triggerExplosion();
    showToast("飲み過ぎ注意！", "warning");
  }

  // 減少で既に超えている場合：穏やかな警告
  if (newLimit < oldLimit && count.value > newLimit) {
    showToast("既に目標を超えています", "info");
  }

  pendingLimit.value = null;
}

// 確認ダイアログで「いいえ」選択時
function handleCancel(): void {
  showConfirmDialog.value = false;
  pendingLimit.value = null;
}

// 目標変更の実行
function applyLimitChange(newLimit: number): void {
  setLimit(newLimit);
}
```

### 3.2 useToast (新規)

トースト表示を管理するComposable

#### 3.2.1 ファイル

- `src/composables/useToast.ts`

#### 3.2.2 インターフェース

```typescript
interface ToastState {
  show: boolean;
  message: string;
  type: "success" | "warning" | "info";
}

interface UseToast {
  toastState: Ref<ToastState>;
  showToast: (
    message: string,
    type: ToastState["type"],
    duration?: number,
  ) => void;
  hideToast: () => void;
}
```

#### 3.2.3 実装

```typescript
import { ref } from "vue";

const toastState = ref<ToastState>({
  show: false,
  message: "",
  type: "info",
});

let toastTimer: number | null = null;

export function useToast(): UseToast {
  const showToast = (
    message: string,
    type: ToastState["type"],
    duration = 2500,
  ) => {
    // 既存のタイマーをクリア
    if (toastTimer) {
      clearTimeout(toastTimer);
    }

    toastState.value = { show: true, message, type };

    // 自動非表示
    toastTimer = window.setTimeout(() => {
      hideToast();
    }, duration);
  };

  const hideToast = () => {
    toastState.value.show = false;
  };

  return {
    toastState,
    showToast,
    hideToast,
  };
}
```
